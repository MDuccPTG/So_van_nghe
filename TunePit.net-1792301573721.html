/* ================= STORAGE ================= */
const KEY = 'so_van_nghe_v1';
let state = { passHash:null, members:[], sessions:[] } ;

function load() {
 try {
   const raw = localStorage.getItem(KEY);
   if(raw) {
     const d = JSON.parse(raw);
     if(d && typeof d === 'object') {
       state.passHash = d.passHash ?? null;
       state.members  = Array.isArray(d.members)  ? d.members  : [];
       state.sessions = Array.isArray(d.sessions) ? d.sessions : [];
      }
    }
  } catch(e) { console.warn('Load lỗi', e); }
}
function save() {
 try { localStorage.setItem(KEY, JSON.stringify(state)); }
 catch(e) { alert('Không lưu được dữ liệu: ' + e.message); }
}
function hash(s) {
 let h = 5381;
 for(let i=0;i<s.length;i++) { h = ((h<<5)+h) + s.charCodeAt(i); h |= 0; }
 return String(h>>>0);
}
const uid = () => Date.now().toString(36) + Math.random().toString(36).slice(2,7);
const esc = s => String(s??'').replace(/[&<>"']/g, c => ( { '&':'&','<':'<','>':'>','"':'"',"'":''' } [c]));
const $ = id => document.getElementById(id);

function todayStr() {
 const d = new Date();
 return d.getFullYear()+'-'+String(d.getMonth()+1).padStart(2,'0')+'-'+String(d.getDate()).padStart(2,'0');
}
function fmtDate(s) {
 if(!s) return '';
 const [y,m,d] = s.split('-');
 return `$ { d } /$ { m } /$ { y } `;
}

/* ================= LOCK ================= */
load();

function showLock() {
 $('app').classList.add('hidden');
 $('lock').classList.remove('hidden');
 const firstTime = !state.passHash;
 $('lockHint').textContent = firstTime ? 'Tạo mật khẩu lần đầu (chỉ mình bạn biết)' : 'Nhập mật khẩu để vào';
 $('lockPass2').classList.toggle('hidden', !firstTime);
 $('lockBtn').textContent = firstTime ? 'Tạo mật khẩu' : 'Vào';
 $('lockPass').value = '';
 $('lockPass2').value = '';
 $('lockErr').textContent = '';
 setTimeout(()=>$('lockPass').focus(), 100);
}

$('lockBtn').onclick = () => {
 const p1 = $('lockPass').value;
 const p2 = $('lockPass2').value;
 const err = $('lockErr');

 if(!p1) { err.textContent = 'Nhập mật khẩu đã.'; return; }

 if(!state.passHash) {
   if(p1.length < 4) { err.textContent = 'Mật khẩu ít nhất 4 ký tự.'; return; }
   if(p1 !== p2) { err.textContent = 'Nhập lại mật khẩu không khớp.'; return; }
   state.passHash = hash(p1);
   save();
   unlock();
  } else {
   if(hash(p1) !== state.passHash) { err.textContent = 'Sai mật khẩu ❌'; return; }
   unlock();
  }
} ;
$('lockPass').onkeydown = e => { if(e.key === 'Enter') $('lockBtn').click(); } ;
$('lockPass2').onkeydown = e => { if(e.key === 'Enter') $('lockBtn').click(); } ;

function unlock() {
 $('lock').classList.add('hidden');
 $('app').classList.remove('hidden');
 openSessionId = null;
 setTab('members');
}

$('btnLock').onclick = () => { showLock(); } ;

document.querySelectorAll('.tabs button').forEach(b => {

function setTab(t) {
 currentTab = t;
 openSessionId = null;
 document.querySelectorAll('.tabs button').forEach(b => b.classList.toggle('on', b.dataset.tab === t));
 render();
}

function render() {
 if(openSessionId) return renderAttendance();
 if(currentTab === 'members')  return renderMembers();
 if(currentTab === 'sessions') return renderSessions();
 if(currentTab === 'stats')    return renderStats();
 if(currentTab === 'settings') return renderSettings();
}

   </div>

     </div>
   </div>`;

 const inp = $('newMember');
 const add = () => {
   const v = inp.value.trim();
   if(!v) return;
   state.members.push( { id: uid(), name: v } );
   save(); renderMembers();
   setTimeout(()=>$('newMember').focus(), 30);
  } ;

 inp.onkeydown = e => { if(e.key === 'Enter') add(); } ;

 m.querySelectorAll('[data-del]').forEach(b => {
   b.onclick = () => {
     const id = b.dataset.del;
     const mem = state.members.find(x=>x.id===id);
     if(!confirm(`Xoá "$ { mem.name } " khỏi danh sách?`)) return;
     state.members = state.members.filter(x=>x.id!==id);
     save(); renderMembers();
    } ;
  } );

   el.onclick = () => {
     const id = el.dataset.rename;
     const mem = state.members.find(x=>x.id===id);
     const v = prompt('Đổi tên thành viên:', mem.name);
     if(v === null) return;
     const nv = v.trim();
     if(!nv) return;
     mem.name = nv; save(); renderMembers();
    } ;
  } );
}

    } ;

/* ================= ATTENDANCE ================= */
function getRec(session, mid) {
 if(!session.records) session.records = { } ;
 if(!session.records[mid]) session.records[mid] = { status:null, attitude:0, note:'' } ;
 return session.records[mid];
}

   stars.forEach(st => {
     st.onclick = () => {
       const v = +st.dataset.v;
       rec.attitude = (rec.attitude === v) ? 0 : v;
       stars.forEach(x => x.classList.toggle('filled', +x.dataset.v <= rec.attitude));
       save();
      } ;
    } );

   const note = row.querySelector('.noteInput');
   note.oninput = () => { rec.note = note.value; save(); } ;
  } );
}

/* ================= STATS ================= */
function renderStats() {
 const m = $('main');
 const totalSess = state.sessions.length;

 if(!state.members.length) {
   m.innerHTML = `<div class="card"><div class="empty">Chưa có dữ liệu.<br>Thêm thành viên và tạo buổi tập trước đã.</div></div>`;
   return;
  }

 const rows = state.members.map(mm => {
   let p=0,l=0,a=0,e=0,attSum=0,attN=0;
   state.sessions.forEach(s => {
     const r = s.records && s.records[mm.id];
     if(!r) return;
     if(r.status === 'present') p++;
     else if(r.status === 'late') l++;
     else if(r.status === 'absent') a++;
     else if(r.status === 'excused') e++;
     if(r.attitude > 0) { attSum += r.attitude; attN++; }
    } );
   const marked = p+l+a+e;
   const rate = marked ? Math.round((p+l)/marked*100) : null;
   const avg = attN ? (attSum/attN) : null;
   return { mm, p, l, a, e, marked, rate, avg } ;
  } );

 rows.sort((x,y) => {
   const rx = x.rate ?? -1, ry = y.rate ?? -1;
   if(ry !== rx) return ry - rx;
   return (y.avg ?? -1) - (x.avg ?? -1);
  } );

 m.innerHTML = `
   <div class="card" style="display:flex;gap:14px;text-align:center">
     <div style="flex:1">
       <div style="font-size:22px;font-weight:900;color:#a855f7">$ { state.members.length } </div>
       <div style="font-size:11.5px;color:#8b8ba7;font-weight:700">THÀNH VIÊN</div>
     </div>
     <div style="width:1px;background:#24263f"></div>
     <div style="flex:1">
       <div style="font-size:22px;font-weight:900;color:#ec4899">$ { totalSess } </div>
       <div style="font-size:11.5px;color:#8b8ba7;font-weight:700">BUỔI TẬP</div>
     </div>
   </div>

   $ { rows.map((r,i) => `
     <div class="statCard">
       <div class="statTop">
         <div class="rank $ { i===0?'g1':i===1?'g2':i===2?'g3':'' } ">$ { i+1 } </div>
         <div class="nm">$ { esc(r.mm.name) } </div>
         <div class="rate">$ { r.rate === null ? '—' : r.rate + '%' } </div>
       </div>
       <div class="mini">
         <span class="p">✓ $ { r.p } </span>
         <span class="l">⏰ $ { r.l } </span>
         <span class="a">✕ $ { r.a } </span>
         <span class="e">📝 $ { r.e } </span>
         <span>Ý thức: $ { r.avg === null ? '—' : r.avg.toFixed(1) + '★' } </span>
       </div>
       <div class="bar"><i style="width:$ { r.rate ?? 0 } %"></i></div>
     </div>
   `).join('') }

   <div class="card" style="background:#141426;border-style:dashed">
     <div style="font-size:12px;color:#7c7c9c;line-height:1.7">
       Tỉ lệ = (Có mặt + Trễ) / số buổi đã điểm danh.<br>
       Ý thức = điểm sao trung bình bạn chấm (1–5).
     </div>
   </div>`;
}

 const m = $('main');
 m.innerHTML = `

     <button class="btn" id="changePass" style="width:100%">Cập nhật mật khẩu</button>
   </div>

   <div class="card">
     <div class="cardTitle">Sao lưu dữ liệu</div>
     <div style="font-size:12.5px;color:#8b8ba7;line-height:1.7;margin-bottom:12px">
       Dữ liệu chỉ nằm trong trình duyệt này. Nếu xoá lịch sử trình duyệt là mất sạch —
       nên xuất file backup thỉnh thoảng.
     </div>
     <button class="btn gray" id="exportBtn" style="width:100%;margin-bottom:8px">⬇️ Xuất file JSON</button>
     <button class="btn gray" id="importBtn" style="width:100%">⬆️ Nhập từ file JSON</button>
     <input id="importFile" type="file" accept=".json,application/json" style="display:none">
   </div>

   <div class="card" style="border-color:#4a1f2a">
     <div class="cardTitle" style="color:#f87171">Vùng nguy hiểm</div>
     <button class="btn gray" id="wipeBtn" style="width:100%;background:#2a1c26;color:#f87171">
       🗑 Xoá toàn bộ dữ liệu
     </button>
   </div>`;

 $('changePass').onclick = () => {
   const o = $('oldPass').value, n = $('newPass').value;
   if(hash(o) !== state.passHash) { alert('Mật khẩu hiện tại sai ❌'); return; }
   if(n.length < 4) { alert('Mật khẩu mới phải từ 4 ký tự'); return; }
   state.passHash = hash(n); save();
   $('oldPass').value = ''; $('newPass').value = '';
   alert('Đã đổi mật khẩu ✅');
  } ;

 $('exportBtn').onclick = () => {
   const blob = new Blob([JSON.stringify(state, null, 2)], { type:'application/json' } );
   const url = URL.createObjectURL(blob);
   const a = document.createElement('a');
   a.href = url;
   a.download = 'so-van-nghe-' + todayStr() + '.json';
   a.click();
   URL.revokeObjectURL(url);
  } ;

 $('importBtn').onclick = () => $('importFile').click();
 $('importFile').onchange = e => {
   const f = e.target.files[0];
   if(!f) return;
   const reader = new FileReader();
   reader.onload = () => {
     try {
       const d = JSON.parse(reader.result);
       if(!d || !Array.isArray(d.members) || !Array.isArray(d.sessions)) throw new Error('File không đúng định dạng');
       if(!confirm('Nhập dữ liệu sẽ GHI ĐÈ toàn bộ dữ liệu hiện tại. Tiếp tục?')) return;
       state.members = d.members;
       state.sessions = d.sessions;
       if(d.passHash) state.passHash = d.passHash;
       save();
       alert('Nhập xong ✅');
       setTab('members');
      } catch(err) { alert('Lỗi: ' + err.message); }
    } ;
   reader.readAsText(f);
   e.target.value = '';
  } ;

 $('wipeBtn').onclick = () => {
   if(!confirm('XOÁ SẠCH mọi thành viên và buổi tập? Không khôi phục được.')) return;
   if(!confirm('Chắc chắn chứ? Lần cuối đó nha.')) return;
   const ph = state.passHash;
   state = { passHash: ph, members: [], sessions: [] } ;
   save();
   setTab('members');
  } ;
}
